Actually simplify — compute at L1 level using incident overlap:

```python
def compute_cross_l2(L0_unique, series_df):
    """cross-L2: which L2 classes share incidents with which."""
    # Build series → l2_class mapping
    l1_l2 = L0_unique.groupby("l1_template")["l2_class"].first().to_dict()
    
    # For each L1, find all L2 classes that appear on the same (host, day)
    host_day_l2 = L0_unique.groupby(["host", "date"])["l2_class"].apply(set).to_dict()
    
    result = {}
    for (l1_template, source), grp in L0_unique.groupby(["l1_template", "source"]):
        related = {}
        my_l2 = l1_l2.get(l1_template, "")
        for _, row in grp.iterrows():
            key = (row["host"], row["date"])
            others = host_day_l2.get(key, set())
            for other_l2 in others:
                if other_l2 != my_l2:
                    related[other_l2] = related.get(other_l2, 0) + 1
        result[(l1_template, source)] = related
    return result
```

```javascript
// ===== SVG chart helpers =====
function sparklineSVG(data, width=160, height=36, color="#4f46e5") {
  if (!data || data.length < 2) return "";
  const max = Math.max(...data, 1);
  const min = Math.min(...data, 0);
  const range = Math.max(max - min, 1);
  const stepX = width / (data.length - 1);
  const points = data.map((v, i) => `${(i * stepX).toFixed(1)},${(height - ((v - min) / range) * (height - 4)).toFixed(1)}`).join(" ");
  return `<svg width="${width}" height="${height}" viewBox="0 0 ${width} ${height}">
    <polyline points="${points}" fill="none" stroke="${color}" stroke-width="1.5"/>
  </svg>`;
}

function barChartSVG(data, width=160, height=36, barColor="#4f46e5") {
  if (!data || data.length === 0) return "";
  const max = Math.max(...data, 1);
  const barW = Math.max(2, Math.floor(width / data.length) - 1);
  const bars = data.map((v, i) => {
    const h = Math.max(1, (v / max) * (height - 4));
    const x = i * (barW + 1);
    const y = height - h - 2;
    return `<rect x="${x}" y="${y}" width="${barW}" height="${h}" fill="${barColor}" rx="1"/>`;
  }).join("");
  return `<svg width="${width}" height="${height}" viewBox="0 0 ${width} ${height}">${bars}</svg>`;
}

function hbarChartSVG(data, width=160, heightPerBar=14, barColor="#4f46e5") {
  const entries = Object.entries(data).sort((a, b) => b[1] - a[1]);
  if (entries.length === 0) return "";
  const max = Math.max(...entries.map(e => e[1]), 1);
  const h = Math.max(heightPerBar * entries.length, 20);
  const barH = Math.max(4, heightPerBar - 2);
  const bars = entries.map(([k, v], i) => {
    const bw = Math.max(2, (v / max) * (width - 20));
    const y = i * heightPerBar;
    return `<rect x="0" y="${y}" width="${bw}" height="${barH}" fill="${barColor}" rx="1"/>
      <text x="${bw + 2}" y="${y + barH - 2}" font-size="8" fill="#6b7280">${k}</text>`;
  }).join("");
  return `<svg width="${width}" height="${h + 2}" viewBox="0 0 ${width} ${h + 2}">${bars}</svg>`;
}
```
